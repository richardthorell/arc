cmake_minimum_required(VERSION 3.22.1)
project(arc_android_project LANGUAGES CXX)

if(NOT ANDROID OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(aarch64|arm64)$")
    message(FATAL_ERROR "ARC Android project compilation requires an ARM64 Android NDK toolchain")
endif()
foreach(name ARC_ANDROID_PROJECT_ROOT ARC_ANDROID_SDK_ROOT)
    if(NOT DEFINED ${name} OR NOT IS_ABSOLUTE "${${name}}" OR NOT IS_DIRECTORY "${${name}}")
        message(FATAL_ERROR "${name} must be an existing absolute directory")
    endif()
    get_filename_component(${name} "${${name}}" REALPATH)
endforeach()

set(_sdk "${ARC_ANDROID_SDK_ROOT}")
if(NOT EXISTS "${_sdk}/arc-installation.json" OR NOT EXISTS "${_sdk}/lib/cmake/ARC/ARCConfig.cmake")
    message(FATAL_ERROR "Selected ARC SDK is missing its installation manifest or CMake package")
endif()
file(READ "${_sdk}/arc-installation.json" _installation)
string(JSON _sdk_version ERROR_VARIABLE _sdk_error GET "${_installation}" engineVersion)
if(_sdk_error OR NOT _sdk_version)
    message(FATAL_ERROR "Invalid ARC SDK installation engine version")
endif()

file(GLOB _descriptors LIST_DIRECTORIES FALSE "${ARC_ANDROID_PROJECT_ROOT}/*.arcproject")
list(LENGTH _descriptors _descriptor_count)
if(NOT _descriptor_count EQUAL 1)
    message(FATAL_ERROR "Android native compilation requires exactly one .arcproject in the selected project root")
endif()
list(GET _descriptors 0 _project_descriptor)
file(READ "${_project_descriptor}" _descriptor)
string(JSON _format ERROR_VARIABLE _format_error GET "${_descriptor}" format)
string(JSON _project_version ERROR_VARIABLE _version_error GET "${_descriptor}" engineVersion)
string(JSON _guid ERROR_VARIABLE _guid_error GET "${_descriptor}" guid)
if(_format_error OR _version_error OR _guid_error OR
   NOT _format STREQUAL "arc-project" OR NOT _project_version STREQUAL _sdk_version OR
   NOT _guid MATCHES "^[0-9a-fA-F-]+$")
    message(FATAL_ERROR "Android project descriptor is invalid or targets an incompatible ARC SDK")
endif()

set(ARC_ANDROID_PROJECT_GUID "${_guid}")
set(ARC_ANDROID_ENGINE_VERSION "${_sdk_version}")
set(ARC_ANDROID_PROJECT_BUILD TRUE)
set(ARC_REFLECTION_GENERATOR "${_sdk}/tools/project_reflection.py")
if(NOT EXISTS "${ARC_REFLECTION_GENERATOR}")
    message(FATAL_ERROR "The installed ARC SDK is missing the reflection generator")
endif()
include("${_sdk}/lib/cmake/ARC/ARCProjectReflection.cmake")

# The project owns its native build target and source list. Do not guess which
# C++ sources to compile, or run the host platform's desktop module build.
add_subdirectory("${ARC_ANDROID_PROJECT_ROOT}" "${CMAKE_BINARY_DIR}/arc-project" EXCLUDE_FROM_ALL)
if(NOT TARGET arc_android_project_runtime)
    message(FATAL_ERROR "Project must define the arc_android_project_runtime shared library for Android")
endif()
get_target_property(_type arc_android_project_runtime TYPE)
if(NOT _type STREQUAL "SHARED_LIBRARY")
    message(FATAL_ERROR "arc_android_project_runtime must be a shared library")
endif()
